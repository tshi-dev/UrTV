# NEVER disable, on any TV, from any list: Android's own core. Regular expressions, one per line, matched against the
# whole package name. Without some of these Android stops at its boot screen and ADB can't reach it to undo
# (CleanYourTV issue #884: a Mi TV Stick after permissioncontroller, packageinstaller, ext.services and the module
# overlays were turned off). optimize/disable.sh refuses a list that names one; optimize/community.sh drops them first.
android
com\.android\.(systemui|settings|shell|phone|packageinstaller|permissioncontroller|externalstorage|inputdevices|keychain|certinstaller|defcontainer|mtp|storagemanager|location\.fused|bluetooth|nfc|server\..*|networkstack.*|tethering|wifi\..*|ext\..*|modulemetadata|providers\..*)
com\.google\.android\.(permissioncontroller|packageinstaller|ext\.services|ext\.shared|modulemetadata|overlay\..*|networkstack.*|gms|gsf|tv\.settings|tvsettings)
.*webview.*
# Added after a Haier boot loop: the MediaTek TV stack and the Android extension services (the names of Clean Your TV's CORE_RE,
# github.com/Gathin23/cleanyourtv, MIT: optimize/lists/community/LICENSE-CleanYourTV.txt covers the lists)
com\.cltv\.mal
com\.mediatek\.(tis|tv\.service|tv\.agent)
android\.ext\.(services|shared)
